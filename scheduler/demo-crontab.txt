# Demo broad-slate scheduler for nba-scenario-bot.
# Install manually with:
#   crontab scheduler/demo-crontab.txt
# Keep the Mac awake while unattended with:
#   caffeinate -s
#
# Runs are fixed a few times per day, not continuously: late morning after the
# daily slate forms, late afternoon for line movement, early evening before main
# US starts, and late night for West Coast plus settlement audit.
# Each command writes one timestamped stdout/stderr log under logs/.
# The entrypoint forces demo mode for its own process and never sets live gates.

CRON_TZ=America/New_York
REPO=/Users/ellingtonfagan/Downloads/nba-scenario-bot
ENTRYPOINT=$REPO/scheduler/run-demo-cycle.sh
LOG_DIR=$REPO/logs

15 11 * * * mkdir -p $LOG_DIR && $ENTRYPOINT >> $LOG_DIR/demo-cycle-$(date +\%Y\%m\%dT\%H\%M\%S)-1115.log 2>&1
10 16 * * * mkdir -p $LOG_DIR && $ENTRYPOINT >> $LOG_DIR/demo-cycle-$(date +\%Y\%m\%dT\%H\%M\%S)-1610.log 2>&1
40 18 * * * mkdir -p $LOG_DIR && $ENTRYPOINT >> $LOG_DIR/demo-cycle-$(date +\%Y\%m\%dT\%H\%M\%S)-1840.log 2>&1
35 23 * * * mkdir -p $LOG_DIR && $ENTRYPOINT >> $LOG_DIR/demo-cycle-$(date +\%Y\%m\%dT\%H\%M\%S)-2335.log 2>&1
